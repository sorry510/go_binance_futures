package main
import(
 "database/sql";"encoding/json";"fmt";"math";"sort";"time"
 _ "go_binance_futures/bootstrap"
 "github.com/beego/beego/v2/core/config"
 _ "github.com/go-sql-driver/mysql"
)
type B struct{T int64;O,H,L,C,Q float64}
type E struct{S string;Y int;Dir,R1,R4,R12 float64}
type S struct{N int `json:"n"`;Mean1,Mean4,Mean12,Win12 float64;PositiveSymbols12,Symbols int;BySymbol map[string]map[string]float64 `json:"by_symbol"`}
type PW struct{P,W float64}
func wq(bs []B,q float64)float64{
 x:=make([]PW,0,len(bs));tot:=0.0
 for _,b:=range bs{w:=b.Q;if w<=0{continue};p:=(b.H+b.L+b.C)/3;x=append(x,PW{p,w});tot+=w}
 if len(x)==0||tot<=0{return math.NaN()}
 sort.Slice(x,func(i,j int)bool{return x[i].P<x[j].P})
 target:=tot*q;c:=0.0
 for _,v:=range x{c+=v.W;if c>=target{return v.P}}
 return x[len(x)-1].P
}
func summary(es []E,syms []string)S{
 z:=S{N:len(es),BySymbol:map[string]map[string]float64{}}
 if len(es)==0{return z}
 for _,x:=range es{z.Mean1+=x.R1;z.Mean4+=x.R4;z.Mean12+=x.R12;if x.R12>0{z.Win12++}}
 z.Mean1/=float64(len(es));z.Mean4/=float64(len(es));z.Mean12/=float64(len(es));z.Win12/=float64(len(es))
 for _,s:=range syms{n:=0;m:=0.0;w:=0;for _,x:=range es{if x.S==s{n++;m+=x.R12;if x.R12>0{w++}}};if n>0{m/=float64(n);z.BySymbol[s]=map[string]float64{"n":float64(n),"mean12":m,"win12":float64(w)/float64(n)};z.Symbols++;if m>0{z.PositiveSymbols12++}}}
 return z
}
func main(){
 syms:=[]string{"BTCUSDT","ETHUSDT","BNBUSDT","XRPUSDT","SOLUSDT","DOGEUSDT","LTCUSDT","AVAXUSDT","UNIUSDT","ZECUSDT"}
 u,_:=config.String("database::username");pw,_:=config.String("database::password");h,_:=config.String("database::host");pt,_:=config.String("database::port");dbn,_:=config.String("database::dbname");if dbn!="go_bn_test"{panic(dbn)}
 db,e:=sql.Open("mysql",fmt.Sprintf("%s:%s@tcp(%s:%s)/%s?charset=utf8mb4",u,pw,h,pt,dbn));if e!=nil{panic(e)};defer db.Close()
 all:=[]E{}
 for _,s:=range syms{
  rows,e:=db.Query("SELECT open_time,open_price,high_price,low_price,close_price,quote_volume FROM market_klines_1h WHERE market='futures_usdt' AND symbol=? ORDER BY open_time",s);if e!=nil{panic(e)}
  b:=[]B{};for rows.Next(){var x B;rows.Scan(&x.T,&x.O,&x.H,&x.L,&x.C,&x.Q);b=append(b,x)};rows.Close()
  if len(b)<100{fmt.Println("SKIP",s,len(b));continue}
  armed:=true;n:=0
  for i:=24;i+12<len(b);i++{
   hist:=b[i-24:i];val:=wq(hist,.15);vah:=wq(hist,.85);if math.IsNaN(val)||math.IsNaN(vah)||val>=vah{continue}
   if b[i].C>=val&&b[i].C<=vah{armed=true;continue}
   if !armed{continue}
   dir:=0.0;if b[i].C>vah{dir=1}else if b[i].C<val{dir=-1}else{continue}
   entry:=b[i+1].O;if entry<=0{continue}
   r1:=dir*math.Log(b[i+1].C/entry);r4:=dir*math.Log(b[i+4].C/entry);r12:=dir*math.Log(b[i+12].C/entry)
   y:=time.UnixMilli(b[i].T).UTC().Year();if y>=2023{all=append(all,E{s,y,dir,r1,r4,r12})}
   armed=false;n++
  }
  fmt.Println("SYMBOL",s,"bars",len(b),"events",n)
 }
 disc:=[]E{};o1:=[]E{};o2:=[]E{}
 for _,x:=range all{if x.Y==2023||x.Y==2024{disc=append(disc,x)};if x.Y==2025{o1=append(o1,x)};if x.Y==2026{o2=append(o2,x)}}
 out:=map[string]S{"discovery_2023_2024":summary(disc,syms),"oos1_2025":summary(o1,syms),"oos2_2026":summary(o2,syms),"all":summary(all,syms)}
 raw,_:=json.MarshalIndent(out,"","  ");fmt.Println(string(raw))
}
