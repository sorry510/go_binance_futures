type Bar struct{T int64;O,C float64}
type E struct{S string;Y int;Z,D,R1,R4,R12 float64}
type EI struct{Symbols []struct{Symbol string `json:"symbol"`;Onboard int64 `json:"onboardDate"`} `json:"symbols"`}
type Sum struct{N int `json:"n"`;Mean1,Win1,Mean4,Win4,Mean12,Win12 float64;PositiveSymbols12,Symbols int;BySymbol map[string]map[string]float64 `json:"by_symbol"`}
func meanSD(x []float64)(float64,float64){s:=0.0;for _,v:=range x{s+=v};m:=s/float64(len(x));q:=0.0;for _,v:=range x{d:=v-m;q+=d*d};return m,math.Sqrt(q/float64(len(x)))}
func avg(e []E,f func(E)float64)(float64,float64){if len(e)==0{return 0,0};s:=0.0;w:=0;for _,x:=range e{v:=f(x);s+=v;if v>0{w++}};return s/float64(len(e)),float64(w)/float64(len(e))}
func summary(e []E,syms []string)Sum{z:=Sum{N:len(e),BySymbol:map[string]map[string]float64{}};z.Mean1,z.Win1=avg(e,func(x E)float64{return x.R1});z.Mean4,z.Win4=avg(e,func(x E)float64{return x.R4});z.Mean12,z.Win12=avg(e,func(x E)float64{return x.R12});for _,s:=range syms{q:=[]E{};for _,x:=range e{if x.S==s{q=append(q,x)}};if len(q)>0{m,w:=avg(q,func(x E)float64{return x.R12});z.BySymbol[s]=map[string]float64{"n":float64(len(q)),"mean12":m,"win12":w};z.Symbols++;if m>0{z.PositiveSymbols12++}}};return z}
func loadBybit(sym string)map[int64]Bar{f,e:=os.Open("/tmp/bybit_1h/"+sym+".csv");if e!=nil{panic(e)};defer f.Close();r:=csv.NewReader(f);r.Read();m:=map[int64]Bar{};for{row,e:=r.Read();if e!=nil{break};t,_:=strconv.ParseInt(row[0],10,64);o,_:=strconv.ParseFloat(row[1],64);c,_:=strconv.ParseFloat(row[2],64);m[t]=Bar{t,o,c}};return m}

 syms:=[]string{"BTCUSDT","ETHUSDT","BNBUSDT","XRPUSDT","SOLUSDT","DOGEUSDT","LTCUSDT","AVAXUSDT","UNIUSDT","ZECUSDT"}

 u,_:=config.String("database::username");pw,_:=config.String("database::password");h,_:=config.String("database::host");pt,_:=config.String("database::port");dbn,_:=config.String("database::dbname");if dbn!="go_bn_test"{panic(dbn)}
 db,e:=sql.Open("mysql",fmt.Sprintf("%s:%s@tcp(%s:%s)/%s?charset=utf8mb4",u,pw,h,pt,dbn));if e!=nil{panic(e)};defer db.Close()

  bin:=[]Bar{};for rows.Next(){var b Bar;rows.Scan(&b.T,&b.O,&b.C);bin=append(bin,b)};rows.Close()

   rb:=math.Log(bin[i].C/bin[i-1].C);ry:=math.Log(b1.C/b0.C);d[i]=ry-rb;valid[i]=true

  }
