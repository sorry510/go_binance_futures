package main
import(
 "database/sql"
 "fmt"
 "math"
 _ "go_binance_futures/bootstrap"
 "github.com/beego/beego/v2/core/config"
 _ "github.com/go-sql-driver/mysql"
)
type B struct{ T int64; O,C float64 }
type E struct{ S string; R1,R4,R12 float64 }
func corr(x,y []float64) float64 {
 n:=len(x); if n==0{return 0}
 mx,my:=0.0,0.0
 for i:=0;i<n;i++{mx+=x[i];my+=y[i]}
 mx/=float64(n);my/=float64(n)
 num,dx,dy:=0.0,0.0,0.0
 for i:=0;i<n;i++{a:=x[i]-mx;b:=y[i]-my;num+=a*b;dx+=a*a;dy+=b*b}
 if dx<=0||dy<=0{return 0}; return num/math.Sqrt(dx*dy)
}
func main(){
 syms:=[]string{"BTCUSDT","ETHUSDT","BNBUSDT","XRPUSDT"}
 u,_:=config.String("database::username");pw,_:=config.String("database::password")
 h,_:=config.String("database::host");pt,_:=config.String("database::port");dbn,_:=config.String("database::dbname")
 if dbn!="go_bn_test"{panic(dbn)}
 db,err:=sql.Open("mysql",fmt.Sprintf("%s:%s@tcp(%s:%s)/%s?charset=utf8mb4",u,pw,h,pt,dbn));if err!=nil{panic(err)};defer db.Close()
 all:=[]E{}
 for _,s:=range syms{
  rows,err:=db.Query("SELECT open_time,open_price,close_price FROM market_klines_1h WHERE market='futures_usdt' AND symbol=? AND open_time>=? AND open_time<? ORDER BY open_time",s,int64(1672531200000),int64(1735689600000));if err!=nil{panic(err)}
  b:=[]B{};for rows.Next(){var x B;rows.Scan(&x.T,&x.O,&x.C);b=append(b,x)};rows.Close()
  sq:=make([]float64,len(b));for i:=1;i<len(b);i++{r:=math.Log(b[i].C/b[i-1].C);sq[i]=r*r}
  ac:=make([]float64,len(b));ok:=make([]bool,len(b))
  for i:=48;i<len(b);i++{ac[i]=corr(sq[i-47:i],sq[i-46:i+1]);ok[i]=true}
  vals:=[]E{}
  for i:=49;i+12<len(b);i++{
   if !ok[i]||!ok[i-1]{continue}
   mom:=b[i].C-b[i-4].C;if mom==0{continue};sgn:=1.0;if mom<0{sgn=-1}
   d:=0.0;if ac[i]>0&&ac[i-1]<=0{d=sgn};if ac[i]<0&&ac[i-1]>=0{d=-sgn};if d==0{continue}
   ent:=b[i+1].O;r1:=d*math.Log(b[i+1].C/ent);r4:=d*math.Log(b[i+4].C/ent);r12:=d*math.Log(b[i+12].C/ent)
   e:=E{s,r1,r4,r12};vals=append(vals,e);all=append(all,e)
  }
  sum:=0.0;w:=0;for _,e:=range vals{sum+=e.R12;if e.R12>0{w++}}
  fmt.Printf("%s n=%d mean12=%.6f win12=%.3f\n",s,len(vals),sum/float64(len(vals)),float64(w)/float64(len(vals)))
 }
 for _,k:=range []int{1,4,12}{sum:=0.0;w:=0;for _,e:=range all{v:=e.R1;if k==4{v=e.R4};if k==12{v=e.R12};sum+=v;if v>0{w++}};fmt.Printf("ALL r%d n=%d mean=%.6f win=%.3f\n",k,len(all),sum/float64(len(all)),float64(w)/float64(len(all)))}
}
