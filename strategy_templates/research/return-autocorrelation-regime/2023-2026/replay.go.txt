func ac1(x []float64)float64{if len(x)<3{return 0};m:=0.0;for _,v:=range x{m+=v};m/=float64(len(x));num,den:=0.0,0.0;for i,v:=range x{d:=v-m;den+=d*d;if i>0{num+=(v-m)*(x[i-1]-m)}};if den<=0{return 0};return num/den}
func summary(es []E,syms []string)S{z:=S{N:len(es),BySymbol:map[string]map[string]float64{}};if len(es)==0{return z};for _,x:=range es{z.Mean1+=x.R1;z.Mean4+=x.R4;z.Mean12+=x.R12;if x.R12>0{z.Win12++}};z.Mean1/=float64(len(es));z.Mean4/=float64(len(es));z.Mean12/=float64(len(es));z.Win12/=float64(len(es));for _,s:=range syms{n:=0;m:=0.0;w:=0;for _,x:=range es{if x.S==s{n++;m+=x.R12;if x.R12>0{w++}}};if n>0{m/=float64(n);z.BySymbol[s]=map[string]float64{"n":float64(n),"mean12":m,"win12":float64(w)/float64(n)};z.Symbols++;if m>0{z.PositiveSymbols12++}}};return z}
