package main
import("database/sql";"fmt";"math";_ "go_binance_futures/bootstrap";"github.com/beego/beego/v2/core/config";_ "github.com/go-sql-driver/mysql")
type B struct{T int64;O,H,L,C float64}
type E struct{S string;R1,R4,R12 float64}
func main(){syms:=[]string{"BTCUSDT","ETHUSDT","BNBUSDT","XRPUSDT"};u,_:=config.String("database::username");pw,_:=config.String("database::password");h,_:=config.String("database::host");pt,_:=config.String("database::port");dbn,_:=config.String("database::dbname");if dbn!="go_bn_test"{panic(dbn)};db,e:=sql.Open("mysql",fmt.Sprintf("%s:%s@tcp(%s:%s)/%s?charset=utf8mb4",u,pw,h,pt,dbn));if e!=nil{panic(e)};defer db.Close();all:=[]E{}
for _,s:=range syms{rows,e:=db.Query("SELECT open_time,open_price,high_price,low_price,close_price FROM market_klines_1h WHERE market='futures_usdt' AND symbol=? AND open_time>=? AND open_time<? ORDER BY open_time",s,int64(1672531200000),int64(1735689600000));if e!=nil{panic(e)};b:=[]B{};for rows.Next(){var x B;rows.Scan(&x.T,&x.O,&x.H,&x.L,&x.C);b=append(b,x)};rows.Close();pv:=make([]float64,len(b));cv:=make([]float64,len(b));for i:=1;i<len(b);i++{if b[i].H>0&&b[i].L>0&&b[i].C>0&&b[i-1].C>0{z:=math.Log(b[i].H/b[i].L);pv[i]=z*z/(4*math.Log(2));r:=math.Log(b[i].C/b[i-1].C);cv[i]=r*r}}
q:=make([]float64,len(b));valid:=make([]bool,len(b));for i:=24;i<len(b);i++{ps,cs:=0.0,0.0;for j:=i-23;j<=i;j++{ps+=pv[j];cs+=cv[j]};if cs>0{q[i]=ps/cs;valid[i]=true}}
n:=0;sum:=0.0;w:=0
for i:=25;i+12<len(b);i++{if !valid[i]||!valid[i-1]{continue};mom:=b[i].C-b[i-4].C;if mom==0{continue};sgn:=1.0;if mom<0{sgn=-1};d:=0.0;if q[i]>1&&q[i-1]<=1{d=-sgn};if q[i]<1&&q[i-1]>=1{d=sgn};if d==0{continue};entry:=b[i+1].O;if entry<=0{continue};r1:=d*math.Log(b[i+1].C/entry);r4:=d*math.Log(b[i+4].C/entry);r12:=d*math.Log(b[i+12].C/entry);all=append(all,E{s,r1,r4,r12});n++;sum+=r12;if r12>0{w++}}
fmt.Printf("%s n=%d mean12=%.6f win12=%.3f\n",s,n,sum/float64(n),float64(w)/float64(n))}
for _,k:=range []int{1,4,12}{sum:=0.0;w:=0;for _,x:=range all{v:=x.R1;if k==4{v=x.R4};if k==12{v=x.R12};sum+=v;if v>0{w++}};fmt.Printf("ALL r%d n=%d mean=%.6f win=%.3f\n",k,len(all),sum/float64(len(all)),float64(w)/float64(len(all)))}
}
