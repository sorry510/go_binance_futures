type Bar struct{T int64; O,C float64}
type Ev struct{S string; Y int; H,M,R1,R4,R12 float64}
type Sum struct{N int `json:"n"`;Mean1,Win1,Mean4,Win4,Mean12,Win12 float64;PositiveSymbols12 int `json:"positive_symbols_12h"`;Symbols int `json:"symbols"`;BySymbol map[string]map[string]float64 `json:"by_symbol"`}
func pe(x []float64)float64{
 c:=[6]int{}; total:=0
 for i:=0;i+2<len(x);i++{
  a:=[]struct{v float64;i int}{{x[i],0},{x[i+1],1},{x[i+2],2}}
  sort.SliceStable(a,func(i,j int)bool{if a[i].v==a[j].v{return a[i].i<a[j].i};return a[i].v<a[j].v})
  p:=[]int{a[0].i,a[1].i,a[2].i}; idx:=0
  switch fmt.Sprint(p){case "[0 1 2]":idx=0;case "[0 2 1]":idx=1;case "[1 0 2]":idx=2;case "[1 2 0]":idx=3;case "[2 0 1]":idx=4;case "[2 1 0]":idx=5}
  c[idx]++;total++
 }
 if total==0{return 1}
 h:=0.0
 for _,n:=range c{if n>0{p:=float64(n)/float64(total);h-=p*math.Log(p)}}
 return h/math.Log(6)
}
func avg(e []Ev,f func(Ev)float64)(float64,float64){if len(e)==0{return 0,0};s:=0.0;w:=0;for _,x:=range e{v:=f(x);s+=v;if v>0{w++}};return s/float64(len(e)),float64(w)/float64(len(e))}
func summary(e []Ev,syms []string)Sum{
 z:=Sum{N:len(e),BySymbol:map[string]map[string]float64{}}
 z.Mean1,z.Win1=avg(e,func(x Ev)float64{return x.R1});z.Mean4,z.Win4=avg(e,func(x Ev)float64{return x.R4});z.Mean12,z.Win12=avg(e,func(x Ev)float64{return x.R12})
 for _,s:=range syms{q:=[]Ev{};for _,x:=range e{if x.S==s{q=append(q,x)}};if len(q)>0{m,w:=avg(q,func(x Ev)float64{return x.R12});z.BySymbol[s]=map[string]float64{"n":float64(len(q)),"mean12":m,"win12":w};z.Symbols++;if m>0{z.PositiveSymbols12++}}}
 return z
}
